cmake_minimum_required(VERSION 3.21.1)
project(PipeLangSqlite VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(PipeLangModel 0.1.0 EXACT CONFIG REQUIRED)
if(NOT UNIX)
    message(FATAL_ERROR "The current SQLite capability requires a qualified POSIX platform")
endif()
if(NOT PIPELANG_SQLITE_MODEL_MANIFEST_SHA256 MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "Select the exact installed PipeLangModel manifest SHA256")
endif()
file(SHA256 "${PipeLangModel_MANIFEST}" actual_model_hash)
if(NOT actual_model_hash STREQUAL PIPELANG_SQLITE_MODEL_MANIFEST_SHA256)
    message(FATAL_ERROR "PipeLangModel manifest checksum differs")
endif()
foreach(source sqlite3.c sqlite3.h)
    if(NOT EXISTS "${PIPELANG_SQLITE_SOURCE_DIRECTORY}/${source}")
        message(FATAL_ERROR "Select the verified SQLite amalgamation directory")
    endif()
endforeach()
file(SHA256 "${PIPELANG_SQLITE_SOURCE_DIRECTORY}/sqlite3.c" sqlite_source_hash)
file(SHA256 "${PIPELANG_SQLITE_SOURCE_DIRECTORY}/sqlite3.h" sqlite_header_hash)
if(NOT sqlite_source_hash STREQUAL "0c657666d819f261cc4007526ad57e1669de9949df7a8ea847aa284a2bf0ee52" OR
   NOT sqlite_header_hash STREQUAL "d6c84557261893439b341d2e6c35187ae3ea24fbfd843ad94616f72d89242b8c")
    message(FATAL_ERROR "SQLite 3.54.1 amalgamation checksum differs")
endif()
add_library(pipelang_sqlite_vendor OBJECT "${PIPELANG_SQLITE_SOURCE_DIRECTORY}/sqlite3.c")
set(PIPELANG_SQLITE_OPTIMIZATION "O3" CACHE STRING "Explicit qualified native optimization profile (O1 or O3)")
if(NOT PIPELANG_SQLITE_OPTIMIZATION MATCHES "^O[13]$")
    message(FATAL_ERROR "Select an explicit O1 or O3 native optimization profile")
endif()
target_compile_definitions(pipelang_sqlite_vendor PRIVATE SQLITE_THREADSAFE=1 SQLITE_ENABLE_FTS5=1
    SQLITE_OMIT_LOAD_EXTENSION=1 SQLITE_OMIT_SHARED_CACHE=1 SQLITE_DQS=0 SQLITE_ENABLE_API_ARMOR=1
    SQLITE_MAX_WORKER_THREADS=0)
target_compile_options(pipelang_sqlite_vendor PRIVATE "-${PIPELANG_SQLITE_OPTIMIZATION}" -g0)
set_target_properties(pipelang_sqlite_vendor PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(pipelang_sqlite STATIC src/Connection.cpp src/Operation.cpp src/Statement.cpp
    $<TARGET_OBJECTS:pipelang_sqlite_vendor>)
add_library(PipeLang::Sqlite ALIAS pipelang_sqlite)
set_target_properties(pipelang_sqlite PROPERTIES EXPORT_NAME Sqlite POSITION_INDEPENDENT_CODE ON)
target_compile_features(pipelang_sqlite PUBLIC cxx_std_20)
target_compile_options(pipelang_sqlite PRIVATE -Wall -Wextra -Wpedantic -Werror "-${PIPELANG_SQLITE_OPTIMIZATION}" -g0)
target_include_directories(pipelang_sqlite PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    PRIVATE "${PIPELANG_SQLITE_SOURCE_DIRECTORY}")
target_link_libraries(pipelang_sqlite PUBLIC PipeLang::ModelRuntime PRIVATE m)
option(PIPELANG_SQLITE_SANITIZERS "Qualify the C++ SQLite boundary with ASan/UBSan" OFF)
if(PIPELANG_SQLITE_SANITIZERS)
    target_compile_options(pipelang_sqlite PRIVATE -O1 -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(pipelang_sqlite PUBLIC -fsanitize=address,undefined)
endif()
add_executable(sqlite-native-tests tests/Main.cpp tests/Functional.cpp tests/Adversarial.cpp tests/Backup.cpp)
target_compile_options(sqlite-native-tests PRIVATE -Wall -Wextra -Wpedantic -Werror "-${PIPELANG_SQLITE_OPTIMIZATION}" -g0)
target_link_libraries(sqlite-native-tests PRIVATE PipeLang::Sqlite)
if(PIPELANG_SQLITE_SANITIZERS)
    target_compile_options(sqlite-native-tests PRIVATE -O1 -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
install(TARGETS pipelang_sqlite EXPORT PipeLangSqliteTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE-SQLite.txt README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/pipelang-sqlite)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../LICENSE" DESTINATION ${CMAKE_INSTALL_DATADIR}/pipelang-sqlite RENAME LICENSE-Apache-2.0)
configure_package_config_file(cmake/PipeLangSqliteConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/PipeLangSqliteConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PipeLangSqlite)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/PipeLangSqliteConfigVersion.cmake"
    VERSION 0.1.0 COMPATIBILITY ExactVersion)
install(EXPORT PipeLangSqliteTargets NAMESPACE PipeLang:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PipeLangSqlite)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/PipeLangSqliteConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/PipeLangSqliteConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PipeLangSqlite)
