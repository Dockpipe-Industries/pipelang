cmake_minimum_required(VERSION 3.20)
project(pipelang_native_model_view_compiler LANGUAGES CXX)
set(PIPELANG_JSON_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/nlohmann/json.hpp")
if(NOT EXISTS "${PIPELANG_JSON_HEADER}")
    message(FATAL_ERROR "Missing pinned nlohmann/json.hpp")
endif()
file(SHA256 "${PIPELANG_JSON_HEADER}" PIPELANG_JSON_SHA256)
if(NOT PIPELANG_JSON_SHA256 STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
    message(FATAL_ERROR "Pinned nlohmann/json.hpp checksum mismatch")
endif()
add_executable(pipelang-model-viewc main.cpp compiler.cpp emitter.cpp ../lexer/lexer.cpp)
target_compile_features(pipelang-model-viewc PRIVATE cxx_std_20)
set_target_properties(pipelang-model-viewc PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(pipelang-model-viewc PRIVATE ../third_party)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(pipelang-model-viewc PRIVATE -Wall -Wextra -Wpedantic)
endif()
