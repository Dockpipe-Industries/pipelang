cmake_minimum_required(VERSION 3.22)
project(PipeLangQtBindingCompiler VERSION 0.1.0 LANGUAGES CXX)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/nlohmann/json.hpp" json_hash)
if(NOT json_hash STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
    message(FATAL_ERROR "Pinned JSON dependency changed")
endif()
add_library(pipelang_qt_binding_generator STATIC BindingModel.cpp Emitter.cpp)
target_compile_features(pipelang_qt_binding_generator PUBLIC cxx_std_20)
add_executable(pipelang-qt-bindc main.cpp)
target_link_libraries(pipelang-qt-bindc PRIVATE pipelang_qt_binding_generator)
add_executable(pipelang-qt-binding-tests Tests.cpp)
target_link_libraries(pipelang-qt-binding-tests PRIVATE pipelang_qt_binding_generator)
foreach(target pipelang_qt_binding_generator pipelang-qt-bindc pipelang-qt-binding-tests)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
enable_testing()
add_test(NAME QtBindingMetadata COMMAND pipelang-qt-binding-tests)
install(TARGETS pipelang-qt-bindc RUNTIME DESTINATION bin)
