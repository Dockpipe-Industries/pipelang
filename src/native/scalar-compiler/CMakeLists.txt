add_library(pipelang_scalar_compiler STATIC compiler.cpp)
target_link_libraries(pipelang_scalar_compiler PUBLIC pipelang_model_compiler pipelang_tool_support)
target_compile_features(pipelang_scalar_compiler PUBLIC cxx_std_20)
add_executable(pipelang-native-cpp main.cpp)
target_link_libraries(pipelang-native-cpp PRIVATE pipelang_scalar_compiler pipelang_sdk_tools)
foreach(target pipelang_scalar_compiler pipelang-native-cpp)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
