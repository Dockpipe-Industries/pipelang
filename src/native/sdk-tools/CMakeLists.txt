    target_link_libraries(pipelang_tool_support PUBLIC advapi32 psapi)
    target_compile_definitions(pipelang_tool_support PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00 NOMINMAX)

    endif()
endforeach()
