cmake_minimum_required(VERSION 3.20)
project(pipelang_cpp_view_parser LANGUAGES CXX)

set(PIPELANG_JSON_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/nlohmann/json.hpp")
if(NOT EXISTS "${PIPELANG_JSON_HEADER}")
    message(FATAL_ERROR "Missing pinned nlohmann/json.hpp")
endif()
file(SHA256 "${PIPELANG_JSON_HEADER}" PIPELANG_JSON_SHA256)
if(NOT PIPELANG_JSON_SHA256 STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
    message(FATAL_ERROR "Pinned nlohmann/json.hpp checksum mismatch")
endif()

add_executable(pipelang-cpp-view-parser main.cpp parser.cpp protocol.cpp)
target_compile_features(pipelang-cpp-view-parser PRIVATE cxx_std_17)
set_target_properties(pipelang-cpp-view-parser PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(pipelang-cpp-view-parser PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../third_party")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(pipelang-cpp-view-parser PRIVATE -Wall -Wextra -Wpedantic)
endif()
