cmake_minimum_required(VERSION 3.20)
project(pipelang_generated_model_view_tests LANGUAGES CXX)
find_package(ForgePipe 0.9.0 EXACT CONFIG REQUIRED COMPONENTS UI)
if(NOT PIPELANG_MODEL_RUNTIME_INCLUDE)
    message(FATAL_ERROR "Explicit installed PipeLang model runtime include is required")
endif()
add_executable(model-view-generated-test GeneratedTest.cpp)
target_compile_features(model-view-generated-test PRIVATE cxx_std_20)
target_include_directories(model-view-generated-test PRIVATE "${PIPELANG_MODEL_RUNTIME_INCLUDE}")
target_link_libraries(model-view-generated-test PRIVATE ForgePipe::UI)
find_package(Threads REQUIRED)
target_link_libraries(model-view-generated-test PRIVATE Threads::Threads)
enable_testing()
add_test(NAME ModelViewGenerated COMMAND model-view-generated-test)
set_tests_properties(ModelViewGenerated PROPERTIES TIMEOUT 10)
